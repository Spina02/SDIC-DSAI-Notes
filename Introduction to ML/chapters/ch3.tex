\chapter{Unsupervised Learning}

\section{Introduction}

In supervised learning, we are given examples $(x, y)$ and the goal is to learn a mapping to predict $y$ from $x$. In \textbf{unsupervised learning}, there are no labels $y$. We only observe the data $X$ and we want to find interesting structures within it.

\begin{definitionblock}[Unsupervised Learning]
\bfit{Unsupervised Learning} is a type of machine learning that looks for previously undetected patterns in a dataset with no pre-existing labels and with a minimum of human supervision.
\end{definitionblock}

Since we do not have a ``ground truth'' to compare against, the goal is often to \textbf{discover} knowledge rather than to learn a specific predictor.
\begin{itemize}
    \item The input is a dataset $D \in \mathcal{P}^*(X)$.
    \item The output is a \textbf{pattern} or structure describing the data.
\end{itemize}

\section{Clustering}

The most common form of unsupervised learning is \textbf{clustering}. We assume that the system generated the data following some grouping scheme, but we do not know the groups.

\begin{definitionblock}[Clustering]
\bfit{Clustering} is the task of grouping a set of objects in such a way that objects in the same group (called a \bfit{cluster}) are more similar to each other than to those in other groups.
\end{definitionblock}

Formally, given a dataset $D \in \mathcal{P}^*(X)$, we want to find a partitioning $\mathcal{C} = \{D_1, \dots, D_k\}$ such that:
\begin{itemize}
    \item $\bigcup_{i=1}^k D_i = D$ (the union covers all data);
    \item $D_i \cap D_j = \emptyset$ for $i \neq j$ (clusters do not overlap);
    \item Elements within $D_i$ are ``close'' to each other;
    \item Elements in different clusters $D_i$, $D_j$ are ``far'' from each other.
\end{itemize}

\subsubsection{Clustering as Optimization}

Clustering can be viewed as an optimization problem where we try to maximize distinction between groups. We want to maximize \textbf{inter-cluster distance} (separation) and minimize \textbf{intra-cluster distance} (compactness). Let $d: X \times X \to \mathbb{R}^+$ be a distance metric. The objective function is:
\small{
$$
\argmax{D_1, \dots, D_k} \left( \left( \sum_{i \neq j} \sum_{x \in D_i, x' \in D_j} d(x, x') \right) - \left( \sum_{i} \sum_{x, x' \in D_i} d(x, x') \right) \right)
$$
}

\vspace{-0.4em}
subject to $\bigcup_{i=1}^k D_i = D$ and $D_i \cap D_j = \emptyset$ for $i \neq j$.

However, finding the global optimum is often computationally unfeasible. Furthermore, the number of clusters $k$ cannot simply be optimized without constraints, because trivial solutions exist:
\begin{itemize}
    \item If $k=1$, compactness is minimized (bad), but separation is undefined.
    \item If $k=n$ (every point is a cluster), compactness is perfect (0), but separation is uninformative.
\end{itemize}
Therefore, $k$ is usually a hyperparameter or determined by heuristics.

\subsubsection{Assessing Clustering}

Since we often lack labels, assessment is difficult.
\begin{enumerate}[leftmargin=*, noitemsep]
    \item \textbf{Manual Inspection}: Visualize the data (e.g., using PCA) and verify if groups make sense.
    \item \textbf{Extrinsic Evaluation}: Use clusters in a downstream task and check if performance improves.
    \item \textbf{Intrinsic Evaluation}: Measure mathematical properties of clusters (compactness, separation).
\end{enumerate}

\subsubsection{The Silhouette Index}

The \textbf{Silhouette Index} measures how similar an object is to its own cluster compared to others.

\begin{definitionblock}[Silhouette Score]
    For a single data point $x \in D_i$ (where $D_i$ is the cluster $x$ belongs to):

    \medskip
\begin{minipage}{0.56\textwidth}
\begin{enumerate}[leftmargin=*, noitemsep]
    \item \textbf{mean intra-cluster distance} $a(x)$:
    
    average distance between $x$ and all other points in $D_i$.
    $$ a(x) = \frac{1}{|D_i| - 1} \sum_{x' \in D_i, x' \neq x} d(x, x') $$
    \item \textbf{mean nearest-cluster distance} $b(x)$: 
    
    average distance between $x$ and all points in the nearest neighboring cluster.
    $$ b(x) = \min_{j \neq i} \frac{1}{|D_j|} \sum_{x' \in D_j} d(x, x') $$
    \item The \textbf{silhouette} $s(x)$ is:
    $$ s(x) = \frac{b(x) - a(x)}{\max\{a(x), b(x)\}} $$
\end{enumerate}
\end{minipage}%
\hfill
\begin{minipage}{0.42\textwidth}
    \centering
    \includegraphics[width=\textwidth]{assets/silhouette.png}
    % \vspace{-0.5em}
    % \captionof{figure}{Silhouette index interpretation.}
\end{minipage}

\medskip

The global Silhouette Score is the mean of $s(x)$ over all observations: $S = \frac{1}{n}\sum_{x} s(x)$.

\end{definitionblock}


Interpretation of $s(x) \in [-1, 1]$:
\begin{itemize}
    \item \textbf{Close to 1}:
    The point is well-clustered (far from neighbors, close to its own group).
    \item \textbf{Close to 0}: 
    The point is on the boundary between two clusters.
    \item \textbf{Close to $-1$}:
    The point is likely assigned to the wrong cluster.
\end{itemize}

\section{Hierarchical Clustering}

Hierarchical clustering builds a clusters' hierarchy, often represented as a tree (or \textbf{dendrogram}). It doesn't require specifying $k$ in advance; cutting the tree at different depths yields different partitions.

\begin{definitionblock}[Hierarchical Clustering Approaches]
\begin{itemize}
    \item \textbf{Agglomerative (Bottom-Up)}: Start with $n$ clusters (each point is a cluster). At each step, merge the two closest clusters until only one remains.
    \item \textbf{Divisive (Top-Down)}: Start with 1 cluster (containing all points). At each step, split a cluster until $n$ clusters remain.
\end{itemize}
\end{definitionblock}

\subsection{Agglomerative Clustering Algorithm}

\begin{algorithm}[H]
    \caption{Agglomerative Hierarchical Clustering}
    \begin{algorithmic}[1]
        \State \textbf{Input:} Dataset $X = \{x^{(1)}, x^{(2)}, \dots, x^{(n)}\}$, distance metric $d(\cdot, \cdot)$, linkage criterion
        \State \textbf{Output:} Dendrogram (hierarchy of clusters)
        \State \textbf{Initialize:} Assign each observation to its own cluster: $\mathcal{C} = \{\{x^{(1)}\}, \{x^{(2)}\}, \dots, \{x^{(n)}\}\}$
        \State \textbf{Compute distance matrix:} Calculate pairwise distances $D_{ij} = d(C_i, C_j)$ for all cluster pairs
        \While{$|\mathcal{C}| > 1$}
            \State Find the pair of clusters $(C_i, C_j)$ with minimum distance:  $(i^*, j^*) = \arg \min_{i < j} D_{ij}$
            \State \textbf{Merge:} Create new cluster $C_{i^*} \cup C_{j^*}$
            \State \textbf{Update:} Remove $C_{i^*}$ and $C_{j^*}$ from $\mathcal{C}$, add $C_{\text{new}}$
            \State \textbf{Update distance matrix:} Compute distances from $C_{\text{new}}$ to all remaining clusters

            \State \textbf{Record merge:} Store the merge in the dendrogram structure
        \EndWhile

        \State \Return the dendrogram representing the complete hierarchy
    \end{algorithmic}
\end{algorithm}

\textbf{Complexity:} $O(n^3)$ for naive implementation; $O(n^2 \log n)$ with efficient data structures.

\subsubsection{Linkage Methods (Cluster Distance)}

The distance between two clusters $D_i$ and $D_j$ is defined by the \textbf{linkage criterion}, which strongly affects the shape of the resulting clusters.

\begin{itemize}
    \item \textbf{Single Linkage}: Distance between the \textbf{closest} pair of points.
    $$ d(D_i, D_j) = \min_{x \in D_i, x' \in D_j} d(x, x') $$
    \textit{Effect:} Can produce long, ``chain-like'' clusters.

    \item \textbf{Complete Linkage}: Distance between the \textbf{farthest} pair of points.
    $$ d(D_i, D_j) = \max_{x \in D_i, x' \in D_j} d(x, x') $$
    \textit{Effect:} Tends to produce compact, spherical clusters.

    \item \textbf{Average Linkage}: Average distance between all pairs.
    $$ d(D_i, D_j) = \frac{1}{|D_i| \cdot |D_j|} \sum_{x \in D_i} \sum_{x' \in D_j} d(x, x') $$

    \item \textbf{Centroid Linkage}: Distance between the centroids (means) of the clusters.
    $$ d(D_i, D_j) = d(\mu_i, \mu_j) $$
\end{itemize}

\section{k-Means Clustering}

k-Means is a \textbf{partitional} clustering algorithm. Unlike hierarchical methods, it requires the number of clusters $k$ to be specified in advance. It attempts to separate samples in $k$ groups of equal variance, minimizing a criterion known as \textbf{inertia} or within-cluster sum-of-squares (WCSS).

\begin{definitionblock}[k-Means Algorithm]
Given a set of observations $(x^{(1)}, x^{(2)}, \dots, x^{(n)})$, partition the $n$ observations into $k$ sets $S = \{S_1, S_2, \dots, S_k\}$ so as to minimize the within-cluster sum of squares. The algorithm is as follows:

\begin{enumerate}[leftmargin=*, noitemsep]
    \item \textbf{Initialization}: Choose $k$ initial centroids $\{\mu_1, \dots, \mu_k\}$ (randomly or using strategies as k-means++).
    \item \textbf{Assignment}: Assign each observation to the cluster with the nearest mean.
    \small{
    $$ S_i^{(t)} = \{x^{(j)} : \|x^{(j)} - \mu_i^{(t)}\|^2 \le \|x^{(j)} - \mu_{l}^{(t)}\|^2 \;\forall l\} $$
    }
    \item \textbf{Update}: Recalculate centroids as the mean of observations assigned to the cluster.
    \small{
    $$ \mu_i^{(t+1)} = \frac{1}{|S_i^{(t)}|} \sum_{x^{(j)} \in S_i^{(t)}} x^{(j)} $$
    }
    \item \textbf{Repeat} steps 2 and 3 until convergence (centroids do not change).
\end{enumerate}
\end{definitionblock}

\begin{itemize}
    \item \textbf{Pros:} Efficient ($O(t \cdot k \cdot n)$ per iteration), simple to implement.
    \item \textbf{Cons:} Requires to set $k$; sensitive to initialization (local minima) and outliers; assumes spherical clusters.
\end{itemize}

\begin{tipsblock}[Sensitivity to Initialization]
k-Means is not deterministic, due to the random initialization of the centroids. It is sensitive to the initial centroids, so it is better to run it multiple times and choose the result with the lowest inertia.
\end{tipsblock}
